Basic Usage

examples/basic-usage/index.ts

Create a framework-neutral API runtime and expose it through Fastify without moving application handlers into Fastify request/reply code.

const runtime = createApiRuntime({
  definition: {
    id: "health-api",
    origin: "internal",
    protocol: "rest",
    basePath: "/api",
    endpoints: {
      health: {
        id: "health",
        kind: "http",
        operations: {
          "health.read": {
            id: "health.read",
            protocol: "http",
            intent: "read",
            method: "GET",
            path: "/health",
          },
        },
      },
    },
  },
  handlers: {
    "health.read": () => ({ body: { ok: true } }),
  },
});
